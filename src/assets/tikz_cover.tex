% ==========================================
    % Constraint Network (Factor Graph) Centerpiece
    % ==========================================
    \begin{center}
    \begin{tikzpicture}[
        scale=1.3, transform shape,
        % Variable Nodes (X_i)
        vnode/.style={
            circle, draw=exocolor, fill=boxbg, 
            text=textcolor, thick, inner sep=0pt, 
            minimum size=9mm, font=\small\bfseries
        },
        % Constraint Factor Nodes (C_j)
        cnode/.style={
            rectangle, rounded corners=3pt, 
            draw=defcolor, fill=boxbg, text=defcolor, 
            thick, inner sep=4pt, font=\scriptsize\bfseries
        },
        % Domain Tag Labels
        dtag/.style={font=\tiny\ttfamily, color=fallbackcolor},
        % Connection Edges
        cedge/.style={draw=textcolor!60, thick}
    ]

        % --- Variable Nodes ---
        \node[vnode] (X1) at (-2.8, 0) {$X_1$};
        \node[vnode] (X2) at (0, 1.6) {$X_2$};
        \node[vnode] (X3) at (0, -1.6) {$X_3$};
        \node[vnode] (X4) at (2.8, 0) {$X_4$};

        % --- Domain Badges (Post-AC Filtered Domains) ---
        \node[dtag, left=2pt] at (X1.west) {$D_1=\{1,2,3\}$};
        \node[dtag, above=2pt] at (X2.north) {$D_2=\{3,\dots,8\}$};
        \node[dtag, below=2pt] at (X3.south) {$D_3=\{2,4,6\}$};
        \node[dtag, right=2pt] at (X4.east) {$D_4=\{4,6,8\}$};

        % --- Constraint Factor Nodes ---
        \node[cnode] (C1) at (-1.4, -0.9) {$2X_1 = X_3$};
        \node[cnode] (C2) at (-1.4, 0.9) {$X_1 < X_2$};
        \node[cnode] (C3) at (1.4, -0.9) {$X_3 + X_4 = 10$};
        \node[cnode] (C4) at (0, 0) {$X_3 < X_2$};

        % --- Network Edges ---
        % C1: 2*X1 = X3
        \draw[cedge] (X1) -- (C1);
        \draw[cedge] (C1) -- (X3);

        % C2: X1 < X2
        \draw[cedge] (X1) -- (C2);
        \draw[cedge] (C2) -- (X2);

        % C3: X3 + X4 = 10
        \draw[cedge] (X3) -- (C3);
        \draw[cedge] (C3) -- (X4);

        % C4: X3 < X2
        \draw[cedge] (X3) -- (C4);
        \draw[cedge] (C4) -- (X2);

    \end{tikzpicture}
    \end{center}
